<!-- Direct Browser ES Module Import via jsDelivr -->
<script type="module">
  import { calculateEMI, calculateAmortization } from "https://cdn.jsdelivr.net/npm/@finengine/math@0.3.0/+esm";
  import { formatBDT } from "https://cdn.jsdelivr.net/npm/@finengine/core@0.3.0/+esm";

  const emi = calculateEMI({ principal: 500000, annualRate: 0.09, tenureMonths: 24 });
  console.log("Monthly EMI:", formatBDT(emi));
</script>
